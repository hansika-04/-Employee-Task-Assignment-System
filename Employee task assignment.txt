import javax.swing.*;
import javax.swing.table.DefaultTableModel;
import java.awt.*;
import java.util.*;

public class EmployeeTaskManager extends JFrame {

    static class Employee {
        int id;
        String name, email, role;

        Employee(int id, String name, String email, String role) {
            this.id = id;
            this.name = name;
            this.email = email;
            this.role = role;
        }
    }

    static class Task {
        int id, employeeId;
        String title, description, status;

        Task(int id, String title, String description,
             int employeeId) {
            this.id = id;
            this.title = title;
            this.description = description;
            this.employeeId = employeeId;
            this.status = "ACTIVE";
        }
    }

    ArrayList<Employee> employees = new ArrayList<>();
    ArrayList<Task> tasks = new ArrayList<>();

    int employeeId = 1, taskId = 1;

    DefaultTableModel employeeModel, taskModel;
    JTable employeeTable, taskTable;

    public EmployeeTaskManager() {
        setTitle("Employee Task Management System");
        setSize(1000, 650);
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        JPanel main = new JPanel(new BorderLayout());
        main.setBackground(new Color(240, 244, 250));

        JLabel heading = new JLabel(
            " EMPLOYEE TASK MANAGEMENT SYSTEM",
            JLabel.CENTER
        );
        heading.setFont(new Font("Arial", Font.BOLD, 25));
        heading.setForeground(Color.WHITE);
        heading.setOpaque(true);
        heading.setBackground(new Color(25, 45, 80));
        heading.setPreferredSize(new Dimension(100, 65));

        main.add(heading, BorderLayout.NORTH);

        JTabbedPane tabs = new JTabbedPane();

        tabs.addTab("Dashboard", dashboard());
        tabs.addTab("Employees", employeePanel());
        tabs.addTab("Tasks", taskPanel());

        main.add(tabs, BorderLayout.CENTER);
        add(main);

        setVisible(true);
    }

    JPanel dashboard() {
        JPanel p = new JPanel(new GridLayout(2, 2, 20, 20));
        p.setBorder(BorderFactory.createEmptyBorder(50, 50, 50, 50));

        JButton total = new JButton("Total Employees");
        JButton active = new JButton("Active Tasks");
        JButton completed = new JButton("Completed Tasks");
        JButton overloaded = new JButton("Overloaded Employees");

        JButton[] buttons = {total, active, completed, overloaded};

        for (JButton b : buttons) {
            b.setFont(new Font("Arial", Font.BOLD, 20));
            b.setBackground(new Color(40, 90, 160));
            b.setForeground(Color.WHITE);
            p.add(b);
        }

        total.addActionListener(e ->
            show("Total Employees: " + employees.size()));

        active.addActionListener(e ->
            show("Active Tasks: " + countStatus("ACTIVE")));

        completed.addActionListener(e ->
            show("Completed Tasks: " + countStatus("COMPLETED")));

        overloaded.addActionListener(e ->
            show("Overloaded Employees: " + overloadedCount()));

        return p;
    }

    JPanel employeePanel() {
        JPanel p = new JPanel(new BorderLayout());

        employeeModel = new DefaultTableModel(
            new String[]{"ID", "Name", "Email", "Role", "Active Tasks"},
            0
        );

        employeeTable = new JTable(employeeModel);

        JPanel form = new JPanel(new FlowLayout());

        JTextField name = new JTextField(10);
        JTextField email = new JTextField(12);
        JTextField role = new JTextField(10);

        JButton add = new JButton("Add Employee");
        JButton edit = new JButton("Edit");
        JButton delete = new JButton("Delete");

        form.add(new JLabel("Name"));
        form.add(name);
        form.add(new JLabel("Email"));
        form.add(email);
        form.add(new JLabel("Role"));
        form.add(role);
        form.add(add);
        form.add(edit);
        form.add(delete);

        add.addActionListener(e -> {
            if (name.getText().isBlank() ||
                email.getText().isBlank()) {
                show("Enter employee name and email");
                return;
            }

            employees.add(new Employee(
                employeeId++,
                name.getText(),
                email.getText(),
                role.getText()
            ));

            refresh();
            name.setText("");
            email.setText("");
            role.setText("");
        });

        edit.addActionListener(e -> {
            int row = employeeTable.getSelectedRow();

            if (row < 0) {
                show("Select an employee");
                return;
            }

            Employee emp = employees.get(row);

            String newName = JOptionPane.showInputDialog(
                this, "Enter name", emp.name);

            if (newName != null && !newName.isBlank()) {
                emp.name = newName;
                refresh();
            }
        });

        delete.addActionListener(e -> {
            int row = employeeTable.getSelectedRow();

            if (row < 0) {
                show("Select an employee");
                return;
            }

            Employee emp = employees.get(row);

            boolean hasTasks = tasks.stream()
                .anyMatch(t -> t.employeeId == emp.id);

            if (hasTasks) {
                show("Reassign or delete tasks first");
                return;
            }

            employees.remove(emp);
            refresh();
        });

        p.add(form, BorderLayout.NORTH);
        p.add(new JScrollPane(employeeTable), BorderLayout.CENTER);

        return p;
    }

    JPanel taskPanel() {
        JPanel p = new JPanel(new BorderLayout());

        taskModel = new DefaultTableModel(
            new String[]{"ID", "Task", "Employee", "Status"},
            0
        );

        taskTable = new JTable(taskModel);

        JPanel form = new JPanel(new FlowLayout());

        JTextField title = new JTextField(12);
        JTextField description = new JTextField(12);

        JComboBox<String> employeeBox = new JComboBox<>();
        JComboBox<String> statusBox = new JComboBox<>(
            new String[]{"ACTIVE", "COMPLETED"}
        );

        JButton add = new JButton("Assign Task");
        JButton update = new JButton("Update Status");
        JButton delete = new JButton("Delete Task");

        form.add(new JLabel("Task"));
        form.add(title);
        form.add(new JLabel("Description"));
        form.add(description);
        form.add(employeeBox);
        form.add(statusBox);
        form.add(add);
        form.add(update);
        form.add(delete);

        employeeBox.setName("employeeBox");

        employeeBox.addPopupMenuListener(
            new javax.swing.event.PopupMenuListener() {
                public void popupMenuWillBecomeVisible(
                    javax.swing.event.PopupMenuEvent e) {
                    employeeBox.removeAllItems();
                    for (Employee emp : employees) {
                        employeeBox.addItem(emp.id + " - " + emp.name);
                    }
                }
                public void popupMenuWillBecomeInvisible(
                    javax.swing.event.PopupMenuEvent e) {}
                public void popupMenuCanceled(
                    javax.swing.event.PopupMenuEvent e) {}
            }
        );

        add.addActionListener(e -> {
            if (title.getText().isBlank() ||
                employeeBox.getSelectedItem() == null) {
                show("Enter task and select employee");
                return;
            }

            int id = Integer.parseInt(
                employeeBox.getSelectedItem().toString().split(" - ")[0]
            );

            String status = statusBox.getSelectedItem().toString();

            if (status.equals("ACTIVE") &&
                activeCount(id) >= 5) {
                show("Assignment rejected! Employee already has 5 active tasks.");
                return;
            }

            Task task = new Task(
                taskId++,
                title.getText(),
                description.getText(),
                id
            );

            task.status = status;
            tasks.add(task);

            refresh();
            title.setText("");
            description.setText("");
        });

        update.addActionListener(e -> {
            int row = taskTable.getSelectedRow();

            if (row < 0) {
                show("Select a task");
                return;
            }

            Task task = tasks.get(row);

            if (task.status.equals("ACTIVE")) {
                task.status = "COMPLETED";
            } else {
                if (activeCount(task.employeeId) >= 5) {
                    show("Cannot reopen! Employee has 5 active tasks.");
                    return;
                }
                task.status = "ACTIVE";
            }

            refresh();
        });

        delete.addActionListener(e -> {
            int row = taskTable.getSelectedRow();

            if (row < 0) {
                show("Select a task");
                return;
            }

            tasks.remove(row);
            refresh();
        });

        p.add(form, BorderLayout.NORTH);
        p.add(new JScrollPane(taskTable), BorderLayout.CENTER);

        return p;
    }

    int activeCount(int id) {
        int count = 0;

        for (Task t : tasks) {
            if (t.employeeId == id &&
                t.status.equals("ACTIVE")) {
                count++;
            }
        }

        return count;
    }

    int countStatus(String status) {
        int count = 0;

        for (Task t : tasks) {
            if (t.status.equals(status)) count++;
        }

        return count;
    }

    int overloadedCount() {
        int count = 0;

        for (Employee e : employees) {
            if (activeCount(e.id) > 5) count++;
        }

        return count;
    }

    void refresh() {
        employeeModel.setRowCount(0);
        taskModel.setRowCount(0);

        for (Employee e : employees) {
            String warning = activeCount(e.id) > 5
                ? " OVERLOADED!" : "";

            employeeModel.addRow(new Object[]{
                e.id, e.name, e.email, e.role,
                activeCount(e.id) + warning
            });
        }

        for (Task t : tasks) {
            String employeeName = "Unknown";

            for (Employee e : employees) {
                if (e.id == t.employeeId) {
                    employeeName = e.name;
                    break;
                }
            }

            taskModel.addRow(new Object[]{
                t.id, t.title, employeeName, t.status
            });
        }
    }

    void show(String message) {
        JOptionPane.showMessageDialog(this, message);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(
            EmployeeTaskManager::new
        );
    }
}
